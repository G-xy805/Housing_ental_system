#### 3.4.1 一键安装脚本 (install.sh)

```bash
#!/bin/bash

# 房屋租售系统安装脚本
# 使用方式：bash install.sh

set -e

echo "=========================================="
echo "  房屋租售系统 - 一键安装脚本"
echo "=========================================="

# 颜色定义
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m' # No Color

# 日志函数
log_info() {
    echo -e "${GREEN}[INFO]${NC} $1"
}

log_warn() {
    echo -e "${YELLOW}[WARN]${NC} $1"
}

log_error() {
    echo -e "${RED}[ERROR]${NC} $1"
}

# 检查是否以 root 运行
check_root() {
    if [ "$EUID" -ne 0 ]; then
        log_error "请使用 root 权限运行此脚本"
        exit 1
    fi
}

# 检查操作系统
check_os() {
    if [ -f /etc/os-release ]; then
        . /etc/os-release
        OS=$NAME
        log_info "检测到操作系统：$OS"
    else
        log_error "无法检测操作系统"
        exit 1
    fi
}

# 安装系统依赖
install_system_deps() {
    log_info "正在安装系统依赖..."
    
    case $OS in
        *"Ubuntu"*|*"Debian"*)
            apt update
            apt install -y curl git wget unzip
            ;;
        *"CentOS"*|*"Red Hat"*)
            yum install -y curl git wget unzip
            ;;
        *)
            log_warn "未知的操作系统，跳过系统依赖安装"
            ;;
    esac
}

# 安装 Node.js
install_nodejs() {
    log_info "正在安装 Node.js..."
    
    if command -v node &> /dev/null; then
        NODE_VERSION=$(node -v)
        log_info "已安装 Node.js: $NODE_VERSION"
    else
        curl -fsSL https://deb.nodesource.com/setup_18.x | bash -
        apt install -y nodejs
        log_info "Node.js 安装完成"
    fi
}

# 安装 Docker
install_docker() {
    log_info "正在安装 Docker..."
    
    if command -v docker &> /dev/null; then
        DOCKER_VERSION=$(docker --version)
        log_info "已安装 Docker: $DOCKER_VERSION"
    else
        curl -fsSL https://get.docker.com | sh
        log_info "Docker 安装完成"
    fi
}

# 安装 Docker Compose
install_docker_compose() {
    log_info "正在安装 Docker Compose..."
    
    if command -v docker-compose &> /dev/null; then
        COMPOSE_VERSION=$(docker-compose --version)
        log_info "已安装 Docker Compose: $COMPOSE_VERSION"
    else
        curl -L "https://github.com/docker/compose/releases/download/v2.24.0/docker-compose-$(uname -s)-$(uname -m)" -o /usr/local/bin/docker-compose
        chmod +x /usr/local/bin/docker-compose
        log_info "Docker Compose 安装完成"
    fi
}

# 创建环境变量文件
create_env_file() {
    log_info "正在创建环境变量文件..."
    
    if [ -f .env ]; then
        log_warn ".env 文件已存在，跳过创建"
    else
        cp .env.example .env
        log_info "环境变量文件创建完成，请编辑 .env 文件配置密码等敏感信息"
    fi
}

# 安装后端依赖
install_backend() {
    log_info "正在安装后端依赖..."
    cd backend
    npm install --production
    cd ..
    log_info "后端依赖安装完成"
}

# 安装前端依赖
install_frontend() {
    log_info "正在安装前端依赖..."
    cd frontend
    npm install
    npm run build
    cd ..
    log_info "前端依赖安装和构建完成"
}

# 初始化数据库
init_database() {
    log_info "正在初始化数据库..."
    # Docker Compose 会自动执行
    log_info "数据库初始化完成"
}

# 启动服务
start_services() {
    log_info "正在启动服务..."
    docker-compose up -d
    log_info "服务启动完成"
}

# 显示安装信息
show_info() {
    echo ""
    echo "=========================================="
    echo "  安装完成！"
    echo "=========================================="
    echo ""
    echo "访问地址：http://localhost"
    echo "API 地址：http://localhost:3000"
    echo ""
    echo "默认管理员账号："
    echo "  用户名：admin"
    echo "  密码：Admin@123"
    echo ""
    echo "请记得修改默认密码！"
    echo ""
    echo "服务管理命令："
    echo "  启动：docker-compose up -d"
    echo "  停止：docker-compose down"
    echo "  重启：docker-compose restart"
    echo "  查看日志：docker-compose logs -f"
    echo ""
}

# 主函数
main() {
    check_root
    check_os
    install_system_deps
    install_nodejs
    install_docker
    install_docker_compose
    create_env_file
    install_backend
    install_frontend
    init_database
    start_services
    show_info
}

# 执行主函数
main
```
